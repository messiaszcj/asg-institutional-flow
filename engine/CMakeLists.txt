cmake_minimum_required(VERSION 3.20)
project(asg_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(asg_engine main.cpp)
if(MSVC)
  target_compile_options(asg_engine PRIVATE /W4 /EHsc)
else()
  target_compile_options(asg_engine PRIVATE -Wall -Wextra)
endif()
